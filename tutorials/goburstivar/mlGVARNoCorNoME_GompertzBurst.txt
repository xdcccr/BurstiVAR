model {
  # Note: mlGVARNoCorNoME_GompertzBurst
  # Trend: Each burst has a constant value following Gompertz curve
  #        trend_b = theta1 * exp(-theta2 * exp(-b * theta3)), where b = burst number (1,2,3,4,5)
  # VAR: Standard VAR(1) process for deviations within each burst
  # Deviation resets at the start of each burst
  
  # PART 1. Level-1 model (within-person)
  for (pp in 1:P) { 
    
    # ==================== Burst 1 (b=1) ====================
    # Trend is constant within burst, computed from Gompertz with b=1
    # First time point of burst 1
    trend[pp,1,1] <- theta[pp,1,1] * exp(-theta[pp,1,2] * exp(-1 * theta[pp,1,3]))
    trend[pp,1,2] <- theta[pp,2,1] * exp(-theta[pp,2,2] * exp(-1 * theta[pp,2,3]))
    mu[pp,1,1:2] <- trend[pp,1,1:2]
    Y[pp,1,1:2] ~ dmnorm(mu[pp,1,1:2], InnovationPrec[1:2,1:2])
    dev[pp,1,1:2] <- Y[pp,1,1:2] - trend[pp,1,1:2]
    
    # Remaining time points in burst 1
    for (tt in 2:T1) {
      trend[pp,tt,1] <- theta[pp,1,1] * exp(-theta[pp,1,2] * exp(-1 * theta[pp,1,3]))
      trend[pp,tt,2] <- theta[pp,2,1] * exp(-theta[pp,2,2] * exp(-1 * theta[pp,2,3]))
      mu[pp,tt,1:2] <- trend[pp,tt,1:2] + c(AR[pp,1]*dev[pp,tt-1,1] + CR[pp,2]*dev[pp,tt-1,2], 
                                            AR[pp,2]*dev[pp,tt-1,2] + CR[pp,1]*dev[pp,tt-1,1])
      Y[pp,tt,1:2] ~ dmnorm(mu[pp,tt,1:2], InnovationPrec[1:2,1:2])
      dev[pp,tt,1:2] <- Y[pp,tt,1:2] - trend[pp,tt,1:2]
    }
    
    # ==================== Burst 2 (b=2) ====================
    # First time point - deviation resets
    trend[pp,T1+1,1] <- theta[pp,1,1] * exp(-theta[pp,1,2] * exp(-2 * theta[pp,1,3]))
    trend[pp,T1+1,2] <- theta[pp,2,1] * exp(-theta[pp,2,2] * exp(-2 * theta[pp,2,3]))
    mu[pp,T1+1,1:2] <- trend[pp,T1+1,1:2]
    Y[pp,T1+1,1:2] ~ dmnorm(mu[pp,T1+1,1:2], InnovationPrec[1:2,1:2])
    dev[pp,T1+1,1:2] <- Y[pp,T1+1,1:2] - trend[pp,T1+1,1:2]
    
    # Remaining time points in burst 2
    for (tt in (T1+2):T2) {
      trend[pp,tt,1] <- theta[pp,1,1] * exp(-theta[pp,1,2] * exp(-2 * theta[pp,1,3]))
      trend[pp,tt,2] <- theta[pp,2,1] * exp(-theta[pp,2,2] * exp(-2 * theta[pp,2,3]))
      mu[pp,tt,1:2] <- trend[pp,tt,1:2] + c(AR[pp,1]*dev[pp,tt-1,1] + CR[pp,2]*dev[pp,tt-1,2], 
                                            AR[pp,2]*dev[pp,tt-1,2] + CR[pp,1]*dev[pp,tt-1,1])
      Y[pp,tt,1:2] ~ dmnorm(mu[pp,tt,1:2], InnovationPrec[1:2,1:2])
      dev[pp,tt,1:2] <- Y[pp,tt,1:2] - trend[pp,tt,1:2]
    }
    
    # ==================== Burst 3 (b=3) ====================
    # First time point - deviation resets
    trend[pp,T2+1,1] <- theta[pp,1,1] * exp(-theta[pp,1,2] * exp(-3 * theta[pp,1,3]))
    trend[pp,T2+1,2] <- theta[pp,2,1] * exp(-theta[pp,2,2] * exp(-3 * theta[pp,2,3]))
    mu[pp,T2+1,1:2] <- trend[pp,T2+1,1:2]
    Y[pp,T2+1,1:2] ~ dmnorm(mu[pp,T2+1,1:2], InnovationPrec[1:2,1:2])
    dev[pp,T2+1,1:2] <- Y[pp,T2+1,1:2] - trend[pp,T2+1,1:2]
    
    # Remaining time points in burst 3
    for (tt in (T2+2):T3) {
      trend[pp,tt,1] <- theta[pp,1,1] * exp(-theta[pp,1,2] * exp(-3 * theta[pp,1,3]))
      trend[pp,tt,2] <- theta[pp,2,1] * exp(-theta[pp,2,2] * exp(-3 * theta[pp,2,3]))
      mu[pp,tt,1:2] <- trend[pp,tt,1:2] + c(AR[pp,1]*dev[pp,tt-1,1] + CR[pp,2]*dev[pp,tt-1,2], 
                                            AR[pp,2]*dev[pp,tt-1,2] + CR[pp,1]*dev[pp,tt-1,1])
      Y[pp,tt,1:2] ~ dmnorm(mu[pp,tt,1:2], InnovationPrec[1:2,1:2])
      dev[pp,tt,1:2] <- Y[pp,tt,1:2] - trend[pp,tt,1:2]
    }
    
    # ==================== Burst 4 (b=4) ====================
    # First time point - deviation resets
    trend[pp,T3+1,1] <- theta[pp,1,1] * exp(-theta[pp,1,2] * exp(-4 * theta[pp,1,3]))
    trend[pp,T3+1,2] <- theta[pp,2,1] * exp(-theta[pp,2,2] * exp(-4 * theta[pp,2,3]))
    mu[pp,T3+1,1:2] <- trend[pp,T3+1,1:2]
    Y[pp,T3+1,1:2] ~ dmnorm(mu[pp,T3+1,1:2], InnovationPrec[1:2,1:2])
    dev[pp,T3+1,1:2] <- Y[pp,T3+1,1:2] - trend[pp,T3+1,1:2]
    
    # Remaining time points in burst 4
    for (tt in (T3+2):T4) {
      trend[pp,tt,1] <- theta[pp,1,1] * exp(-theta[pp,1,2] * exp(-4 * theta[pp,1,3]))
      trend[pp,tt,2] <- theta[pp,2,1] * exp(-theta[pp,2,2] * exp(-4 * theta[pp,2,3]))
      mu[pp,tt,1:2] <- trend[pp,tt,1:2] + c(AR[pp,1]*dev[pp,tt-1,1] + CR[pp,2]*dev[pp,tt-1,2], 
                                            AR[pp,2]*dev[pp,tt-1,2] + CR[pp,1]*dev[pp,tt-1,1])
      Y[pp,tt,1:2] ~ dmnorm(mu[pp,tt,1:2], InnovationPrec[1:2,1:2])
      dev[pp,tt,1:2] <- Y[pp,tt,1:2] - trend[pp,tt,1:2]
    }
    
    # ==================== Burst 5 (b=5) ====================
    # First time point - deviation resets
    trend[pp,T4+1,1] <- theta[pp,1,1] * exp(-theta[pp,1,2] * exp(-5 * theta[pp,1,3]))
    trend[pp,T4+1,2] <- theta[pp,2,1] * exp(-theta[pp,2,2] * exp(-5 * theta[pp,2,3]))
    mu[pp,T4+1,1:2] <- trend[pp,T4+1,1:2]
    Y[pp,T4+1,1:2] ~ dmnorm(mu[pp,T4+1,1:2], InnovationPrec[1:2,1:2])
    dev[pp,T4+1,1:2] <- Y[pp,T4+1,1:2] - trend[pp,T4+1,1:2]
    
    # Remaining time points in burst 5
    for (tt in (T4+2):T5) {
      trend[pp,tt,1] <- theta[pp,1,1] * exp(-theta[pp,1,2] * exp(-5 * theta[pp,1,3]))
      trend[pp,tt,2] <- theta[pp,2,1] * exp(-theta[pp,2,2] * exp(-5 * theta[pp,2,3]))
      mu[pp,tt,1:2] <- trend[pp,tt,1:2] + c(AR[pp,1]*dev[pp,tt-1,1] + CR[pp,2]*dev[pp,tt-1,2], 
                                            AR[pp,2]*dev[pp,tt-1,2] + CR[pp,1]*dev[pp,tt-1,1])
      Y[pp,tt,1:2] ~ dmnorm(mu[pp,tt,1:2], InnovationPrec[1:2,1:2])
      dev[pp,tt,1:2] <- Y[pp,tt,1:2] - trend[pp,tt,1:2]
    }
    
    # ==================== PART 2. Level-2 model (between-person) ====================
    
    # Gompertz parameters (6 parameters: 3 per variable)
    for(k in 1:6){
      person_params[pp,k] ~ dnorm(Level2Mean[k], tau_L2[k])T(0,999)
    }
    
    # AR & CR parameters (4 parameters)
    for(k in 1:4){
      person_params[pp,k+6] ~ dnorm(Level2Mean[k+6], tau_L2[k+6])
    }
    
    # Gompertz parameters mapping
    theta[pp,1,1] <- person_params[pp,1]  # y1 asymptote
    theta[pp,1,2] <- person_params[pp,2]  # y1 displacement
    theta[pp,1,3] <- person_params[pp,3]  # y1 growth rate
    theta[pp,2,1] <- person_params[pp,4]  # y2 asymptote
    theta[pp,2,2] <- person_params[pp,5]  # y2 displacement
    theta[pp,2,3] <- person_params[pp,6]  # y2 growth rate
    
    # VAR parameters mapping
    AR[pp,1] <- person_params[pp,7]   # y1 AR
    AR[pp,2] <- person_params[pp,8]   # y2 AR
    CR[pp,1] <- person_params[pp,9]   # y1->y2
    CR[pp,2] <- person_params[pp,10]  # y2->y1
  }
  
  # ==================== PART 3. Prior distributions ====================
  
  # Gompertz parameter means
  # True values: theta1_y1=10, theta2_y1=3, theta3_y1=1.0
  #              theta1_y2=8,  theta2_y2=2.5, theta3_y2=0.8
  Level2Mean[1] ~ dunif(0, 25)    # y1 asymptote (true: 10)
  Level2Mean[2] ~ dunif(0, 10)    # y1 displacement (true: 3)
  Level2Mean[3] ~ dunif(0, 3)     # y1 growth rate (true: 1.0)
  Level2Mean[4] ~ dunif(0, 25)    # y2 asymptote (true: 8)
  Level2Mean[5] ~ dunif(0, 10)    # y2 displacement (true: 2.5)
  Level2Mean[6] ~ dunif(0, 3)     # y2 growth rate (true: 0.8)
  
  # VAR parameter means (same as GompertzCurve model)
  Level2Mean[7] ~ dnorm(0, 0.01)T(-1.5, 1.5)   # AR1 (true: 0.3)
  Level2Mean[8] ~ dnorm(0, 0.01)T(-1.5, 1.5)   # AR2 (true: 0.2)
  Level2Mean[9] ~ dnorm(0, 0.01)               # CR y1->y2 (true: -0.15)
  Level2Mean[10] ~ dnorm(0, 0.01)              # CR y2->y1 (true: -0.1)
  
  # Gompertz parameter variances
  # True SDs: sigma1_y1=2, sigma2_y1=0.5, sigma3_y1=0.2
  #           sigma1_y2=1.5, sigma2_y2=0.4, sigma3_y2=0.15
  tau_L2[1] ~ dgamma(1, 1)      # y1 asymptote (SD~2)
  tau_L2[2] ~ dgamma(1, 1)      # y1 displacement (SD~0.5)
  tau_L2[3] ~ dgamma(1, 0.01)   # y1 growth rate (SD~0.2, smaller)
  tau_L2[4] ~ dgamma(1, 1)      # y2 asymptote (SD~1.5)
  tau_L2[5] ~ dgamma(1, 1)      # y2 displacement (SD~0.4)
  tau_L2[6] ~ dgamma(1, 0.01)   # y2 growth rate (SD~0.15, smaller)
  
  for(k in 1:6){
    Level2Sigma[k] <- 1/sqrt(tau_L2[k])
  }
  
  # VAR parameter variances (true SD: 0.1)
  for(k in 1:4){
    tau_L2[k+6] ~ dgamma(1, 0.01)
    Level2Sigma[k+6] <- 1/sqrt(tau_L2[k+6])
  }
  
  # Innovation precision matrix
  InnovationPrec[1:2,1:2] ~ dwish(W0[1:2,1:2], df_res)
  sigma_innovation[1:2,1:2] <- inverse(InnovationPrec[1:2,1:2])
  
}
