model {
  # PART 1. Level-1 model (within-person)
  for (pp in 1:P) { 
    # ==================== Burst 1 ====================
    # first time point
    trend[pp,1,1:2] <- InterceptBurst1[pp,1:2]
    mu[pp,1,1:2] <- trend[pp,1,1:2]
    Y[pp,1,1:2] ~ dmnorm(mu[pp,1,1:2], InnovationPrec[1:2,1:2])
    dev[pp,1,1:2] <- Y[pp,1,1:2] - trend[pp,1,1:2] 
    
    # t>=2 Burst1
    for (tt in 2:T1) { 
      # current trend
      trend[pp,tt,1:2] <- InterceptBurst1[pp,1:2]
      mu[pp,tt,1:2] <- trend[pp,tt,1:2]+c(AR[pp,1]*dev[pp,tt-1,1] + CR[pp,2]*dev[pp,tt-1,2], 
                                    AR[pp,2]*dev[pp,tt-1,2] + CR[pp,1]*dev[pp,tt-1,1])
      Y[pp,tt,1:2] ~ dmnorm(mu[pp,tt,1:2],InnovationPrec[1:2,1:2])
      dev[pp,tt,1:2] = Y[pp,tt,1:2] - trend[pp,tt,1:2]
    }
    
    # ==================== Burst 2 ====================
    # first time point
    trend[pp,T1+1,1:2] <- InterceptBurst2[pp,1:2]
    mu[pp,T1+1,1:2] <- trend[pp,T1+1,1:2]
    Y[pp,T1+1,1:2] ~ dmnorm(mu[pp,T1+1,1:2], InnovationPrec[1:2,1:2])
    dev[pp,T1+1,1:2] <- Y[pp,T1+1,1:2] - trend[pp,T1+1,1:2] 
    for (tt in (T1+2):T2) { 
      # current trend
      trend[pp,tt,1:2] <- InterceptBurst2[pp,1:2]
      mu[pp,tt,1:2] <- trend[pp,tt,1:2]+c(AR[pp,1]*dev[pp,tt-1,1] + CR[pp,2]*dev[pp,tt-1,2], 
                                    AR[pp,2]*dev[pp,tt-1,2] + CR[pp,1]*dev[pp,tt-1,1])
      Y[pp,tt,1:2] ~ dmnorm(mu[pp,tt,1:2],InnovationPrec[1:2,1:2])
      dev[pp,tt,1:2] = Y[pp,tt,1:2] - trend[pp,tt,1:2]
    }
    
    # ==================== Burst 3 ====================
    # first time point
    trend[pp,T2+1,1:2] <- InterceptBurst3[pp,1:2]
    mu[pp,T2+1,1:2] <- trend[pp,T2+1,1:2]
    Y[pp,T2+1,1:2] ~ dmnorm(mu[pp,T2+1,1:2], InnovationPrec[1:2,1:2])
    dev[pp,T2+1,1:2] <- Y[pp,T2+1,1:2] - trend[pp,T2+1,1:2] 
    for (tt in (T2+2):T3) { 
      # current trend
      trend[pp,tt,1:2] <- InterceptBurst3[pp,1:2]
      mu[pp,tt,1:2] <- trend[pp,tt,1:2]+c(AR[pp,1]*dev[pp,tt-1,1] + CR[pp,2]*dev[pp,tt-1,2], 
                                    AR[pp,2]*dev[pp,tt-1,2] + CR[pp,1]*dev[pp,tt-1,1])
      Y[pp,tt,1:2] ~ dmnorm(mu[pp,tt,1:2],InnovationPrec[1:2,1:2])
      dev[pp,tt,1:2] = Y[pp,tt,1:2] - trend[pp,tt,1:2]
    }
    
    # ==================== Burst 4 ====================
    # first time point
    trend[pp,T3+1,1:2] <- InterceptBurst4[pp,1:2]
    mu[pp,T3+1,1:2] <- trend[pp,T3+1,1:2]
    Y[pp,T3+1,1:2] ~ dmnorm(mu[pp,T3+1,1:2], InnovationPrec[1:2,1:2])
    dev[pp,T3+1,1:2] <- Y[pp,T3+1,1:2] - trend[pp,T3+1,1:2] 
    for (tt in (T3+2):T4) { 
      # current trend
      trend[pp,tt,1:2] <- InterceptBurst4[pp,1:2]
      mu[pp,tt,1:2] <- trend[pp,tt,1:2]+c(AR[pp,1]*dev[pp,tt-1,1] + CR[pp,2]*dev[pp,tt-1,2], 
                                    AR[pp,2]*dev[pp,tt-1,2] + CR[pp,1]*dev[pp,tt-1,1])
      Y[pp,tt,1:2] ~ dmnorm(mu[pp,tt,1:2],InnovationPrec[1:2,1:2])
      dev[pp,tt,1:2] = Y[pp,tt,1:2] - trend[pp,tt,1:2]
    }
    
    # ==================== Burst 5 ====================
    # first time point
    trend[pp,T4+1,1:2] <- InterceptBurst5[pp,1:2]
    mu[pp,T4+1,1:2] <- trend[pp,T4+1,1:2]
    Y[pp,T4+1,1:2] ~ dmnorm(mu[pp,T4+1,1:2], InnovationPrec[1:2,1:2])
    dev[pp,T4+1,1:2] <- Y[pp,T4+1,1:2] - trend[pp,T4+1,1:2] 
    for (tt in (T4+2):T5) { 
      # current trend
      trend[pp,tt,1:2] <- InterceptBurst5[pp,1:2]
      mu[pp,tt,1:2] <- trend[pp,tt,1:2]+c(AR[pp,1]*dev[pp,tt-1,1] + CR[pp,2]*dev[pp,tt-1,2], 
                                    AR[pp,2]*dev[pp,tt-1,2] + CR[pp,1]*dev[pp,tt-1,1])
      Y[pp,tt,1:2] ~ dmnorm(mu[pp,tt,1:2],InnovationPrec[1:2,1:2])
      dev[pp,tt,1:2] = Y[pp,tt,1:2] - trend[pp,tt,1:2]
    }
    
    # PART 2. Level-2 model (between-person)
    
    # Intercept parameters (10 parameters: 5 bursts x 2 variables)
    for(k in 1:10){
    person_params[pp,k] ~ dnorm(Level2Mean[k], tau_L2[k])
    }
    
    # AR & CR parameters (4 parameters)
    for(k in 11:14){
    person_params[pp,k] ~ dnorm(Level2Mean[k], tau_L2[k])
    }
    
    # Burst Intercepts
    InterceptBurst1[pp,1:2] <- person_params[pp,1:2]
    InterceptBurst2[pp,1:2] <- person_params[pp,3:4]
    InterceptBurst3[pp,1:2] <- person_params[pp,5:6]
    InterceptBurst4[pp,1:2] <- person_params[pp,7:8]
    InterceptBurst5[pp,1:2] <- person_params[pp,9:10]
    
    # VAR parameters
    AR[pp,1] <- person_params[pp,11]   # reading AR
    AR[pp,2] <- person_params[pp,12]   # math AR
    CR[pp,1] <- person_params[pp,13]   # reading->math
    CR[pp,2] <- person_params[pp,14]   # math->reading
  }
  
  # PART 3. Prior distributions
  # Mean for Intercepts (10 parameters)
  Level2Mean[1] ~ dnorm(0,1/25)
  Level2Mean[2] ~ dnorm(0,1/25)
  Level2Mean[3] ~ dnorm(0,1/25)
  Level2Mean[4] ~ dnorm(0,1/25)
  Level2Mean[5] ~ dnorm(0,1/25)
  Level2Mean[6] ~ dnorm(0,1/25)
  Level2Mean[7] ~ dnorm(0,1/25)
  Level2Mean[8] ~ dnorm(0,1/25)
  Level2Mean[9] ~ dnorm(0,1/25)
  Level2Mean[10] ~ dnorm(0,1/25)
  
  # Mean for AR & CR (4 parameters)
  Level2Mean[11] ~ dnorm(0,0.01)T(-1.5,1.5)
  Level2Mean[12] ~ dnorm(0,0.01)T(-1.5,1.5)
  Level2Mean[13] ~ dnorm(0,0.01)
  Level2Mean[14] ~ dnorm(0,0.01)

  
  # Variance for Intercepts
  for(k in 1:10){
  tau_L2[k] ~ dgamma(1, 1)
  Level2Sigma[k] <- 1/sqrt(tau_L2[k])
  }
  
  # Variance for AR & CR
  for(k in 11:14){
  tau_L2[k] ~ dgamma(1, 0.01)
  Level2Sigma[k] <- 1/sqrt(tau_L2[k])
}

  # Innovation precision matrix
  InnovationPrec[1:2,1:2] ~ dwish(W0[1:2,1:2], df_res)
  sigma_innovation[1:2,1:2] <- inverse(InnovationPrec[1:2,1:2])
  
}
